\section{Backwards sweep}

For an operator $L$ we approximate the exponential over short times $\Delta s$
by
\begin{align}
  \exp(L\Delta s) \approx \exp_h(L\Delta s) = (1-\Delta s L/2)^{-1}(1 + \Delta s L/2),\\
   \exp(-L\Delta s) \approx \exp_h(-L\Delta s) = (1+\Delta s L/2)^{-1}(1 - \Delta s L/2),\\
\end{align}
Then, for longer times $m\Delta s$, we approximate
\begin{align}
  \exp(Lm\Delta s) \approx \exp_h(L\Delta s)^m,\\
   \exp(-Lm\Delta s) \approx \exp_h(-L\Delta s)^m.
\end{align}
The averaged nonlinearity is
\begin{equation}
  X^0 = \sum_{m=0}^{M-1} w_m \exp_h(-L\Delta s)^m N(W_m).
\end{equation}
If we introduce
\begin{equation}
  X^k = \sum_{m=k}^{M-1} w_m \exp_h(-L\Delta s)^(m-k) N(W_m),
\end{equation}
then we have the recurrence
\begin{align}
  X^{k-1} & = \sum_{m={k-1}}^{M-1} w_m \exp_h(-L\Delta s)^(m-k+1) N(W_m), \\
  & = \sum_{m={k-1}}^{M-1} w_m \exp_h(-L\Delta s)^(m-k+1) N(W_m), \\
  & = \sum_{m=k^{M-1} w_m \exp_h(-L\Delta s)^(m-k+1) N(W_m)
    + w_{k-1}N(w_{k-1}}, \\
    & = \sum_{m=k^{M-1} w_m \exp_h(-L\Delta s)^(m-k+1) N(W_m)
    + w_{k-1}N(w_{k-1}}, \\
  & = \exp_h(-L\Delta s)\sum_{m=k^{M-1} w_m \exp_h(-L\Delta s)^(m-k) N(W_m)
    + w_{k-1}N(w_{k-1}}, \\
  & = \exp_h(-L\delta s)X^k
    + w_{k-1}N(w_{k-1}),
\end{align}
i.e.,
\begin{equation}
(I + \Delta s L/2)X^{k-1} = (I - \Delta s L/2)\left(X^k + w_{k-1}N(w_{k-1}\right).  
\end{equation}
